# Налаштування прошивки Pi 5 для Motion Logger (власний образ Buildroot)
# Крок 4.7: прошивка запускає U-Boot, а він з boot.scr вантажить Image. Без U-Boot: kernel=Image.
arm_64bit=1
kernel=u-boot.bin
disable_overscan=1

# UART прошивки: лог самої прошивки виходить і на GPIO14 (DBG1, пін 8, TX; видно 15.10.2026). Консоллю DBG1 не є:
# у ядрі UART0 вимкнений (/dev/ttyAMA0 нема), а на нашій Pi пін 10 (GPIO15, RX) замкнений на землю. U-Boot, ядро,
# init і getty працюють на JST-роз'ємі Pi 5 (ttyAMA10, cmdline.txt і BR2_TARGET_GENERIC_GETTY_PORT).
enable_uart=1
enable_rp1_uart=1

# I2C1 (GPIO2/3) для MPU6050 і SPI0 (GPIO7-11) для ADXL345. Дубль навмисно: оверлей motion-logger-sensors
# нижче теж ставить status = "okay" для i2c1 і spi0, але шини критичні для проєкту, тому вмикаємо їх ще й тут,
# незалежно від оверлея (правило Юрія Безкора, 09.11.2026). Що станеться без цих двох рядків, не перевіряли.
# Драйвери зібрано модулями, їх вантажить S11modules зі списку /etc/modules-load.d/motion_logger.conf.
dtparam=i2c_arm=on
dtparam=spi=on

# Датчики: власний оверлей board/motion_logger/dts/motion-logger-sensors.dts (MPU6050 на I2C1, ADXL345 на
# SPI0 CE0, драйвери ядра IIO) і UART2 (GPIO4/5) для GPS. Оверлеї збирає і кладе в образ post-image.sh.
dtoverlay=motion-logger-sensors
dtoverlay=uart2-pi5

# Середній світлодіод (світлодіод 2) висить на GPIO6, а GPIO0-8 після старту прошивки мають підтяжку до
# живлення («ip» у pinctrl): крізь неї йде слабкий струм і світлодіод ледь тліє. Тримаємо лінію виходом із
# низьким рівнем уже з прошивки, ще до запуску motion_logger.
gpio=6=op,dl
